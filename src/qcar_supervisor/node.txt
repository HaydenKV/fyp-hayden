// sup_jmf_node.cpp
// Jump Markov Filter node with IMU + velocity updates only,
// and node‐level hysteresis for mode switching.

#include <ros/ros.h>
#include <sensor_msgs/Imu.h>
#include <sensor_msgs/JointState.h>
#include <nav_msgs/Odometry.h>
#include <std_msgs/Float64MultiArray.h>
#include <std_msgs/Int32.h>
#include <tf/transform_broadcaster.h>
#include <Eigen/Dense>
#include <array>
#include <vector>
#include "qcar_supervisor/JMF/JMF.h"
#include "qcar_visnav/estimation/model/kinematic_model.h"

using qcar_supervisor::JumpMarkovFilter;
using qcar_supervisor::KinematicModel;

class JmfNode {
public:
  JmfNode(ros::NodeHandle& nh, ros::NodeHandle& pnh)
    : nh_(nh)
    , pnh_(pnh)
    , delta_meas_(0.0)
    , declared_mode_(0)
    , stable_count_(0)
    , min_stable_(3)         // require three consecutive cycles
    , p_thresh_(0.80)        // require >80% probability to switch
  {
    // 0) Timing init
    last_time_ = ros::Time::now();
    ROS_INFO_STREAM("[JMF] Node namespace: " << pnh_.getNamespace());

    // 1) Load per‐mode params
    mode_params_.fill(KinematicModel::ModelParams());
    R_imu_modes_.resize(JumpMarkovFilter::M);
    R_vel_modes_.resize(JumpMarkovFilter::M);

    static const std::array<std::string, JumpMarkovFilter::M> names = {
      "healthy", "motor_fault", "steering_fault", "imu_fault", "encoder_fault"
    };

    for(int i = 0; i < JumpMarkovFilter::M; ++i) {
      ros::NodeHandle mnh(pnh_, "mode_params/" + names[i]);

      // model length L
      mnh.param("L", mode_params_[i].L, 0.258);
      ROS_INFO_STREAM("[JMF] Loaded " << names[i] << ".L = "
                      << mode_params_[i].L);

      // process‐noise vector q
      std::vector<double> qv;
      mnh.getParam("q", qv);
      mode_params_[i].q = Eigen::Map<const Eigen::Matrix<double,6,1>>(qv.data());
      ROS_INFO_STREAM("[JMF] " << names[i] << ".q = "
                      << mode_params_[i].q.transpose());

      // R_imu
      std::vector<double> buf;
      mnh.getParam("R_imu", buf);
      R_imu_modes_[i] = Eigen::Map<const Eigen::Matrix2d>(buf.data());
      ROS_INFO_STREAM("[JMF] " << names[i] << ".R_imu =\n"
                      << R_imu_modes_[i]);

      // R_vel
      mnh.getParam("R_vel", buf);
      R_vel_modes_[i] = Eigen::Map<const Eigen::Matrix2d>(buf.data());
      ROS_INFO_STREAM("[JMF] " << names[i] << ".R_vel =\n"
                      << R_vel_modes_[i]);
    }

    // 2) Load π (row-major)
    std::vector<double> pi_flat;
    pnh_.param("pi", pi_flat, std::vector<double>{});
    int N = std::sqrt(pi_flat.size());
    if(N*N != (int)pi_flat.size()) {
      ROS_FATAL_STREAM("[JMF] 'pi' size " << pi_flat.size()
                       << " is not a perfect square");
      ros::shutdown();
      return;
    }
    Eigen::Map<const Eigen::Matrix<double,Eigen::Dynamic,Eigen::Dynamic,
                                   Eigen::RowMajor>> Pi_map(
      pi_flat.data(), N, N);
    ROS_INFO_STREAM("[JMF] Loaded π =\n" << Pi_map);

    // 3) Initialize JMF
    auto x0 = KinematicModel::StateVec::Zero();
    auto P0 = KinematicModel::StateMat::Identity() * 0.1;
    Eigen::VectorXd mode0 = Eigen::VectorXd::Ones(N) / double(N);

    jmf_.setModelParams(mode_params_);
    jmf_.init(x0, P0, mode0, Pi_map);

    // 4) Subscribers
    imu_sub_   = nh_.subscribe("/imu",           10,
                                &JmfNode::imuCallback,    this);
    vel_sub_   = nh_.subscribe("/qcar/ekf/odom", 10,
                                &JmfNode::velCallback,    this);
    joint_sub_ = nh_.subscribe("/joint_states",  10,
                                &JmfNode::jointCallback,  this);

    // 5) Publishers & timer
    fused_pub_ = nh_.advertise<nav_msgs::Odometry>(
                  "sup_jmf/odom",      10);
    modes_pub_ = nh_.advertise<std_msgs::Float64MultiArray>(
                  "sup_jmf/mode_prob", 10);
    mode_pub_  = nh_.advertise<std_msgs::Int32>(
                  "sup_jmf/current_mode", 10);

    pnh_.param("publish_rate", publish_rate_, 50.0);
    publish_timer_ = nh_.createTimer(
      ros::Duration(1.0/publish_rate_),
      &JmfNode::timerPublishCallback, this);

    ROS_INFO("[JMF] Node initialized.");
  }

private:
  // 1) IMU callback: predict + IMU-update
  void imuCallback(const sensor_msgs::Imu::ConstPtr& msg) {
    double dt = (msg->header.stamp - last_time_).toSec();
    if(dt <= 0.0) return;
    last_time_ = msg->header.stamp;

    // Build input
    KinematicModel::ModelInput u;
    u.a_meas_x = msg->linear_acceleration.x;
    u.a_meas_y = msg->linear_acceleration.y;
    u.delta    = delta_meas_;
    u.dt       = dt;

    // 1) predict
    jmf_.setInput(u);
    jmf_.predict(msg->header.stamp.toSec(), dt);

    // 2) IMU measurement model: [a_x - bax; atan(r)]
    auto h = [&](auto const& x){
      Eigen::Vector2d zh;
      zh << x(0) + x(4),
            std::atan(x(2));
      return zh;
    };
    auto H = [&](auto const& x){
      Eigen::Matrix<double,2,6> M; M.setZero();
      M(0,0)=1; M(0,4)=1;
      M(1,2)=1.0/(1.0 + x(2)*x(2));
      return M;
    };

    Eigen::Vector2d z{ msg->linear_acceleration.x,
                       std::atan(delta_meas_) };

    int best = jmf_.mostLikelyMode();
    jmf_.update(z, h, H, R_imu_modes_[best]);
  }

  // 2) Velocity callback: velocity-update
  void velCallback(const nav_msgs::Odometry::ConstPtr& msg) {
    Eigen::Vector2d z{ msg->twist.twist.linear.x,
                       msg->twist.twist.linear.y };

    int best = jmf_.mostLikelyMode();
    auto const& Rv = R_vel_modes_[best];

    auto h = [&](auto const& x){
      Eigen::Vector2d zh; zh << x(0), x(1); return zh;
    };
    auto H = [&](auto const&){
      Eigen::Matrix<double,2,6> M; M.setZero();
      M(0,0)=1; M(1,1)=1;
      return M;
    };

    jmf_.update(z, h, H, Rv);
  }

  // 3) Joint callback: update steering angle
  void jointCallback(const sensor_msgs::JointState::ConstPtr& msg) {
    for(size_t i = 0; i < msg->name.size(); ++i) {
      if(msg->name[i] == "steering_joint") {
        delta_meas_ = msg->position[i];
        break;
      }
    }
  }

  // 4) Publish fused state, mode probs, and declared mode
  void timerPublishCallback(const ros::TimerEvent&) {
    // Fused state → fused odom
    auto xf = jmf_.fusedState();
    nav_msgs::Odometry odom;
    odom.header.stamp    = ros::Time::now();
    odom.header.frame_id = "odom";
    odom.pose.pose.orientation =
      tf::createQuaternionMsgFromYaw(xf(2));
    odom.twist.twist.linear.x  = xf(0);
    odom.twist.twist.linear.y  = xf(1);
    odom.twist.twist.angular.z = xf(2);
    fused_pub_.publish(odom);

    // Raw mode probabilities
    Eigen::VectorXd probs = jmf_.modeProb();
    std_msgs::Float64MultiArray arr;
    arr.data.assign(probs.data(), probs.data() + probs.size());
    modes_pub_.publish(arr);

    // Hysteresis: only switch if new mode > p_thresh_ for min_stable_ cycles
    int best = jmf_.mostLikelyMode();
    double pbest = probs(best);

    if(best != declared_mode_) {
      if(pbest > p_thresh_) {
        if(++stable_count_ >= min_stable_) {
          declared_mode_ = best;
          stable_count_ = 0;
        }
      } else {
        stable_count_ = 0;
      }
    } else {
      stable_count_ = 0;
    }

    std_msgs::Int32 mi; 
    mi.data = declared_mode_;
    mode_pub_.publish(mi);

    // Broadcast TF
    static tf::TransformBroadcaster br;
    tf::Transform t;
    t.setOrigin({xf(0), xf(1), 0.0});
    t.setRotation({
      odom.pose.pose.orientation.x,
      odom.pose.pose.orientation.y,
      odom.pose.pose.orientation.z,
      odom.pose.pose.orientation.w
    });
    br.sendTransform(tf::StampedTransform(t,
                                          odom.header.stamp,
                                          "odom", "base_link"));
  }

  // ── Members ─────────────────────────────────────────────────────────
  ros::NodeHandle    nh_, pnh_;
  ros::Subscriber    imu_sub_, vel_sub_, joint_sub_;
  ros::Publisher     fused_pub_, modes_pub_, mode_pub_;
  ros::Timer         publish_timer_;
  ros::Time          last_time_;
  double             delta_meas_, publish_rate_;

  JumpMarkovFilter   jmf_;
  std::array<KinematicModel::ModelParams,
             JumpMarkovFilter::M> mode_params_;
  std::vector<Eigen::Matrix2d>     R_imu_modes_, R_vel_modes_;

  // hysteresis state
  int    declared_mode_;
  int    stable_count_;
  const int    min_stable_;
  const double p_thresh_;
};

int main(int argc, char** argv) {
  ros::init(argc, argv, "sup_jmf_node");
  ros::NodeHandle nh, pnh("~");
  JmfNode node(nh, pnh);
  ros::spin();
  return 0;
}
